\begin{tikzpicture}[>=Stealth, font=\small, scale=0.9]
  % --- panel A: two point clouds in one behavior plane ---
  \begin{scope}[shift={(0,0)}]
    \draw[->,gray] (-0.4,0) -- (4.6,0) node[right,font=\scriptsize]{$c_1$};
    \draw[->,gray] (0,-0.4) -- (0,4.6) node[above,font=\scriptsize]{$c_2$};
    % real cloud (blue dots, two lobes = two outcome regions)
    \foreach \x/\y in {1.0/1.0, 1.3/1.2, 0.9/1.4, 1.2/0.8, 1.5/1.1, 1.1/1.05, 1.35/0.95,
                       3.2/3.4, 3.5/3.2, 3.1/3.6, 3.4/3.55, 3.6/3.3}
      \fill[blue!70] (\x,\y) circle (1.6pt);
    % sim points (orange diamonds) reaching one lobe, missing the other
    \foreach \x/\y in {1.05/1.1, 1.25/1.25, 1.4/0.9, 0.95/0.95}
      \draw[orange, thick] (\x,\y) node[diamond, fill=orange, inner sep=1.4pt] {};
    \node[font=\scriptsize, blue!70!black] at (1.6,0.45) {real};
    \node[font=\scriptsize, orange!80!black] at (2.5,1.6) {sim};
    \node[font=\scriptsize, anchor=south] at (2.1,4.7) {(a) behavior plane};
  \end{scope}
  % --- panel B: density grid with occupancy threshold ---
  \begin{scope}[shift={(6.2,0)}]
    \draw[black!20, step=0.55] (0,0) grid (3.85,3.85);
    \draw[->,gray] (-0.4,0) -- (4.6,0) node[right,font=\scriptsize]{$c_1$};
    \draw[->,gray] (0,-0.4) -- (0,4.6) node[above,font=\scriptsize]{$c_2$};
    % occupied cells (real density above threshold)
    \foreach \x/\y in {1.1/1.1, 1.65/1.1, 1.1/1.65, 1.65/1.65, 0.55/1.1, 1.1/0.55,
                       3.3/3.3, 3.3/2.75, 2.75/3.3}
      \fill[blue!25, opacity=0.9] (\x-0.275,\y-0.275) rectangle (\x+0.275,\y+0.275);
    % one dense-core marker
    \node[font=\scriptsize, anchor=south] at (2.1,4.7) {(b) occupancy $\rho > \varepsilon \cdot \mathrm{core}$};
  \end{scope}
  % --- panel C: covered vs uncovered ---
  \begin{scope}[shift={(12.4,0)}]
    \draw[black!20, step=0.55] (0,0) grid (3.85,3.85);
    \draw[->,gray] (-0.4,0) -- (4.6,0) node[right,font=\scriptsize]{$c_1$};

    \foreach \x/\y in {1.1/0.55, 3.3/3.3, 3.3/2.75, 2.75/3.3}
      \fill[red!35] (\x-0.275,\y-0.275) rectangle (\x+0.275,\y+0.275);
    % exemplar callout from an uncovered cell to its nearest real conversation
    \draw[->,red!60!black] (3.3,3.0) .. controls (4.4,2.2) .. (4.5,1.2)
      node[below,font=\scriptsize,align=left,anchor=north west,xshift=-3mm]
      {exemplar: nearest real\\conversation (top-$K$)};
    \node[font=\scriptsize, anchor=south] at (2.1,4.7) {(c) coverage vs uncovered};
  \end{scope}
\end{tikzpicture}
