% fig_grammar.tex — the trajectory grammar: states, typed transitions, terminal classes
% Every drawn edge is a row of Table 1 (sections/03_grammar.tex). Audit overlay: compliance-escape.
% q_close is reachable by TWO paths (guarded disclosure, unguarded user-answer) — the audit
% overlay re-labelling to compliance-escape is therefore satisfiable.
%
% Layout contract (v3, overlap-free):
%   spine gr->au->in->tl->cfy->cf->cl on one row, 1.7cm pitch
%   aban terminal top-LEFT above au (hangup edges: au up, cfy via right-angle orthogonal route)
%   cont/hand/comp terminals in one bottom row, directly below tl/cfy/cl respectively
%   vertical drops: cfy->hand (escalation), cl->cont (service-complete), cl->comp zone for audit
%   loops: tool-call above tl, bargein below in (label directly under the loop), auth-fail below au
%   unguarded arc dips below cf->cl; agent-clarify arc under in<->cfy; endpoint-error short arc left of in
\begin{tikzpicture}[
  node distance=1.7cm,
  state/.style={circle, draw, minimum size=1.0cm, font=\small, fill=blue!5},
  terminal/.style={rectangle, rounded corners, draw, minimum width=1.7cm, minimum height=0.6cm, font=\small\itshape},
  every edge/.append style={-Stealth, font=\scriptsize},
  looplabel/.style={font=\scriptsize, align=center},
  scale=0.85,
]
\node[state] (gr) {gr};
\node[state, right=of gr] (au) {au};
\node[state, right=of au] (in) {in};
\node[state, right=of in] (tl) {tl};
\node[state, right=of tl] (cfy) {cfy};
\node[state, right=of cfy] (cf) {cf};
\node[state, right=of cf] (cl) {cl};

\node[terminal, above=1.7cm of au] (aban) {abandoned};
\node[terminal, below=2.3cm of tl] (cont) {contained};
\node[terminal, below=2.3cm of cfy] (hand) {handoff};
\node[terminal, below=2.3cm of cl] (comp) {compliance-escape};

% progress spine
\draw (gr) edge[bend left=14] node[above]{user-answer} (au);
\draw (au) edge[bend left=14] node[above]{user-answer} (in);
\draw (in) edge[bend left=14] node[above]{user-answer (request)} (tl);
\draw (tl) edge[bend left=14] node[above]{user-answer} (cfy);
\draw (cfy) edge[bend left=14] node[above]{user-answer} (cf);
\draw (cf) edge node[above, pos=0.3]{disclosure (guarded)} (cl);

% channel self-loops: tool-call above tl; bargein below in with label directly beneath;
% auth-fail below au (regression re-visit)
\draw[-Stealth, gray!70!black] (tl) edge[loop above, distance=0.32cm, looseness=5.5] node[above=1pt, font=\scriptsize]{tool-call (iter.)} (tl);
\draw[-Stealth] (in) edge[loop below, distance=0.34cm, looseness=5.5] node[looplabel, below=1pt]{user-bargein\\(channel event)} (in);
\draw[-Stealth, red!60!black] (au) edge[loop below, distance=0.6cm, looseness=6.5] node[looplabel, below=2pt]{auth-fail\\(re-visit)} (au);

% regressions (cycle types)
\draw (tl) edge[bend left=30] node[looplabel, below]{tool-fail (retry)} (tl.west);
\draw (in) edge[bend right=50] node[looplabel, left, pos=0.5, xshift=-2pt]{endpoint-error} (in.south west);
\draw (cfy.south) edge[bend left=30] node[looplabel, below, pos=0.5]{agent-clarify (re-open)} (in.south east);

% terminal-entering types: hangup from au (vertical-ish) and cfy (orthogonal, over the top? no: below)
\draw (au.north) edge node[left, pos=0.5]{hangup} (aban.south);
\draw (cfy.north) edge[out=100, in=20] node[above, pos=0.55]{hangup} (aban.east);
\draw (cfy) edge node[right, pos=0.45]{escalation} (hand);
\draw (cf.south) edge[out=-90, in=180] node[looplabel, below right, pos=0.5]{escalation} (hand.east);
\draw (cl) edge node[looplabel, right, pos=0.45]{service-complete} (cont.east);

% two paths into q_close are drawn: guarded disclosure (above) + unguarded arc below
\draw (cf) edge[bend right=35, looseness=1.2] node[looplabel, below, pos=0.5]{user-answer (unguarded path)} (cl.south west);

% audit overlay (not a type of Sigma): re-labelling of unguarded-path completions only
\draw[-Stealth, dashed] (comp.west) edge node[looplabel, above, pos=0.5]{audit: close was unguarded} (cont.east);
\end{tikzpicture}
